cmake_minimum_required(VERSION 3.24)
project(myndhamr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
# Ninja builds invoked by AGP must respect a bounded compiler working set.
set_property(GLOBAL PROPERTY JOB_POOLS myndhamr_compile=4)
set(CMAKE_JOB_POOL_COMPILE myndhamr_compile)
include(CTest)
include(FetchContent)
include(dependencies.cmake)
option(MYNDHAMR_BUILD_JNI "Build JVM/Android binding" ON)
option(MYNDHAMR_BUILD_BENCHMARKS "Build smoke benchmark" ON)
set(protobuf_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(protobuf_INSTALL OFF CACHE BOOL "" FORCE)
set(protobuf_WITH_ZLIB OFF CACHE BOOL "" FORCE)
set(protobuf_BUILD_LIBUPB OFF CACHE BOOL "" FORCE)
set(ABSL_PROPAGATE_CXX_STD ON CACHE BOOL "" FORCE)
set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(absl
  URL https://github.com/abseil/abseil-cpp/releases/download/${MYNDHAMR_ABSEIL_VERSION}/abseil-cpp-${MYNDHAMR_ABSEIL_VERSION}.tar.gz
  URL_HASH SHA256=${MYNDHAMR_ABSEIL_SHA256}
  DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
FetchContent_MakeAvailable(absl)
if(ANDROID)
  if(NOT MYNDHAMR_HOST_PROTOC)
    message(FATAL_ERROR "Set MYNDHAMR_HOST_PROTOC to the host protoc from the documented host build")
  endif()
  set(WITH_PROTOC "${MYNDHAMR_HOST_PROTOC}" CACHE FILEPATH "" FORCE)
endif()
FetchContent_Declare(protobuf
  URL https://github.com/protocolbuffers/protobuf/releases/download/v${MYNDHAMR_PROTOBUF_VERSION}/protobuf-${MYNDHAMR_PROTOBUF_VERSION}.tar.gz
  URL_HASH SHA256=${MYNDHAMR_PROTOBUF_SHA256}
  DOWNLOAD_EXTRACT_TIMESTAMP FALSE)
FetchContent_MakeAvailable(protobuf)
set(PROTO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../shared/scan-format/src/main/proto")
set(PROTO_REL myndhamr/scan/v1/foundation.proto)
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${GENERATED}")
add_custom_command(
  OUTPUT "${GENERATED}/myndhamr/scan/v1/foundation.pb.cc" "${GENERATED}/myndhamr/scan/v1/foundation.pb.h"
  COMMAND protobuf::protoc --proto_path=${PROTO_ROOT} --cpp_out=${GENERATED} ${PROTO_ROOT}/${PROTO_REL}
  DEPENDS protobuf::protoc "${PROTO_ROOT}/${PROTO_REL}"
  VERBATIM)
add_library(myndhamr_schema STATIC "${GENERATED}/myndhamr/scan/v1/foundation.pb.cc")
target_include_directories(myndhamr_schema SYSTEM PUBLIC "${GENERATED}")
target_link_libraries(myndhamr_schema PUBLIC protobuf::libprotobuf)
add_library(myndhamr_core STATIC core/src/foundation.cpp)
target_include_directories(myndhamr_core PUBLIC core/include)
target_link_libraries(myndhamr_core PUBLIC myndhamr_schema)
function(myndhamr_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
myndhamr_warnings(myndhamr_core)
if(MYNDHAMR_BUILD_JNI)
  add_library(myndhamr_jni SHARED bindings/jni.cpp)
  target_link_libraries(myndhamr_jni PRIVATE myndhamr_core)
  if(NOT ANDROID)
    find_package(JNI REQUIRED COMPONENTS JVM)
    target_include_directories(myndhamr_jni PRIVATE ${JNI_INCLUDE_DIRS})
  endif()
  myndhamr_warnings(myndhamr_jni)
endif()
if(BUILD_TESTING)
  add_executable(myndhamr_native_tests tests/foundation_test.cpp)
  target_link_libraries(myndhamr_native_tests PRIVATE myndhamr_core)
  myndhamr_warnings(myndhamr_native_tests)
  add_test(NAME foundation COMMAND myndhamr_native_tests)
endif()
if(MYNDHAMR_BUILD_BENCHMARKS)
  add_executable(myndhamr_benchmark benchmarks/serialization.cpp)
  target_link_libraries(myndhamr_benchmark PRIVATE myndhamr_core)
  myndhamr_warnings(myndhamr_benchmark)
endif()
